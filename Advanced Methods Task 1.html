# create dfm
dfm_claims <- dfm(tokens)

# trim dfm
dfm_claims_1pct <- dfm_claims %>%
  dfm_trim(
    min_docfreq = 0.01,
    docfreq_type = "prop"
  )

dfm_claims_5pct <- dfm_claims %>%
  dfm_trim(
    min_docfreq = 0.05,
    docfreq_type = "prop"
  )

dfm_claims_10pct <- dfm_claims %>%
  dfm_trim(
    min_docfreq = 0.1,
    docfreq_type = "prop"
  )

dfm_claims_15pct <- dfm_claims %>%
  dfm_trim(
    min_docfreq = 0.15,
    docfreq_type = "prop"
  )

# get sparsity

# fix pseudo random number generator
set.seed(17091826)

# split data
train_ids <- claims %>%
  mutate(id = row_number()) %>%
  slice_sample(prop = 0.8) %>%
  pull(id)

test_ids <- setdiff(seq_len(nrow(claims)), train_ids)

train_dfm <- dfm_claims_15pct[train_ids, ]
test_dfm  <- dfm_claims_15pct[test_ids, ]

y_train <- claims$Ingroup[train_ids]
y_test  <- claims$Ingroup[test_ids]

# train naive bayes
naive_bayes <- textmodel_nb(
  x = train_dfm, # X_train
  y = y_train,
  prior = "docfreq"
)

# validate
predictions <- predict(
  naive_bayes,
  newdata = test_dfm, # X_test
  type = "class"
)

# get metrics
confusionMatrix(
  predictions,
  as.factor(y_test)
)
